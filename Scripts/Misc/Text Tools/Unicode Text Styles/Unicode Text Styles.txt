function cp(n) {
    if (n <= 0xFFFF) return String.fromCharCode(n);
    n -= 0x10000;
    return String.fromCharCode(0xD800 + (n >> 10), 0xDC00 + (n & 0x3FF));
}

function mapMath(style, ch) {
    var c = ch.charCodeAt(0);
    var upper = c >= 65 && c <= 90;
    var lower = c >= 97 && c <= 122;
    var digit = c >= 48 && c <= 57;

    if (style === 9) {
        if (upper) return cp(0x1D400 + c - 65);
        if (lower) return cp(0x1D41A + c - 97);
        if (digit) return cp(0x1D7CE + c - 48);
    } else if (style === 10) {
        if (upper) return cp(0x1D434 + c - 65);
        if (lower) return c === 104 ? cp(0x210E) : cp(0x1D44E + c - 97);
    } else if (style === 11) {
        if (upper) return cp(0x1D468 + c - 65);
        if (lower) return cp(0x1D482 + c - 97);
    } else if (style === 12) {
        if (upper) return cp(SCRIPT_U[c] || (0x1D49C + c - 65));
        if (lower) return cp(SCRIPT_L[c] || (0x1D4B6 + c - 97));
    } else if (style === 13) {
        if (upper) return cp(0x1D4D0 + c - 65);
        if (lower) return cp(0x1D4EA + c - 97);
    } else if (style === 14) {
        if (upper) return cp(FRAK_U[c] || (0x1D504 + c - 65));
        if (lower) return cp(0x1D51E + c - 97);
    } else if (style === 15) {
        if (upper) return cp(DS_U[c] || (0x1D538 + c - 65));
        if (lower) return cp(0x1D552 + c - 97);
        if (digit) return cp(0x1D7D8 + c - 48);
    } else if (style === 16) {
        return SUPER[ch] || ch;
    } else {
        return SUB[ch] || ch;
    }
    return ch;
}

function cyrAccent(ch, style) {
    var code = ch.charCodeAt(0);
    var mi = code % CYR_MARKS.length;
    var r = ch + CYR_MARKS[mi];
    if (style === 2) r += CYR_MARKS[(mi + 2) % CYR_MARKS.length];
    return r;
}

var TOTAL = 17;
function mapBase(style, ch, ci) {
    var lower = ch.toLowerCase(), code = lower.charCodeAt(0), n = code - 97 + 1;
    var latin = n >= 1 && n <= 26 && lower.length === 1;
    var cyr = lower.length === 1 && code >= 0x0400 && code <= 0x04FF;
    var upper = ch !== lower, r = ch, v, idx, tr;
    if (style === 1) { if (latin) { v = ALPHA_ACCENT[n - 1]; r = v.charAt(0); if (upper) r = r.toUpperCase(); } else if (cyr) r = cyrAccent(ch, 1); }
    else if (style === 2) { if (latin) { v = ALPHA_HEAVY[n - 1]; r = v.charAt(0); if (upper) r = r.toUpperCase(); } else if (cyr) r = cyrAccent(ch, 2); }
    else if (style === 3) { if (latin) r = SMALL[n - 1]; }
    else if (style === 4) { var c4 = ch.charCodeAt(0); if (c4 >= 33 && c4 <= 126) r = String.fromCharCode(c4 + 65248); else if (ch === " ") r = "\u3000"; }
    else if (style === 5) { if (ch !== " " && ch !== "\n" && ch !== "\r") r = ch + "\u0336"; }
    else if (style === 6) { if (latin) r = LEET[n - 1]; else if (cyr && CYR_LEET[lower]) { r = CYR_LEET[lower]; if (upper) r = r.toUpperCase(); } }
    else if (style === 7) { if (latin) { v = ALPHA_ACCENT[n - 1]; idx = (17 + (ci + 1) * 13 + n * 7) % v.length; r = v.charAt(idx); if (upper) r = r.toUpperCase(); } else if (cyr) { tr = CYR_TRANSLIT[lower]; if (tr !== undefined) { r = tr; if (upper && r) r = r.charAt(0).toUpperCase() + r.slice(1); } } }
    else if (style === 8) { if (ch !== " " && ch !== "\n" && ch !== "\r") r = ch + MARKS[(1 + (ci + 1) * 3 + ch.charCodeAt(0)) % MARKS.length]; }
    return r;
}
function transformOne(str, style) {
    var out = "", chars = (typeof Array.from === "function") ? Array.from(str) : str.split("");
    for (var ci = 0; ci < chars.length; ci++) out += style > 8 ? mapMath(style, chars[ci]) : mapBase(style, chars[ci], ci);
    return out;
}
function transformAll(str, count) {
    count = count || TOTAL; var out = [];
    for (var s = 1; s <= count; s++) out.push(transformOne(str, s));
    return out;
}
