  function detectDominantLang(text) {
    var counts = { ru:0, en:0, zh:0, ja:0, ko:0, ar:0 };
    for (var i = 0; i < text.length; i++) {
      var c = text.charCodeAt(i);
      if ((c >= 0x0430 && c <= 0x044F) || c === 0x0451) counts.ru++;
      else if ((c >= 0x41 && c <= 0x5A) || (c >= 0x61 && c <= 0x7A)) counts.en++;
      else if (c >= 0x4E00 && c <= 0x9FFF) counts.zh++;
      else if ((c >= 0x3040 && c <= 0x309F) || (c >= 0x30A0 && c <= 0x30FF)) counts.ja++;
      else if (c >= 0xAC00 && c <= 0xD7A3) counts.ko++;
      else if (c >= 0x0600 && c <= 0x06FF) counts.ar++;
    }
    var best = 'en', max = 0;
    for (var lang in counts) {
      if (counts[lang] > max) { max = counts[lang]; best = lang; }
    }
    log("Lang detect: ru=" + counts.ru + " en=" + counts.en +
        " zh=" + counts.zh + " ja=" + counts.ja +
        " ko=" + counts.ko + " ar=" + counts.ar +
        " -> " + best, "Lang");
    return best;
  }

  function parseDictVariants(json, pair) {
    var out = [];
    if (!json) return out;
    function addEntry(e) {
      if (!e || !e.tr) return;
      for (var k = 0; k < e.tr.length && out.length < dictMaxVariants; k++) {
        var t = e.tr[k];
        if (!t || !t.text) continue;
        var pos = "";
        if (t.pos) {
          var raw = (typeof t.pos === "string") ? t.pos : (t.pos.text || t.pos.code || "");
          pos = posFullName(raw);
        }
        var backs = [];
        if (t.mean && t.mean.length) {
          for (var m = 0; m < Math.min(5, t.mean.length); m++) {
            if (t.mean[m] && t.mean[m].text && backs.indexOf(t.mean[m].text) === -1) backs.push(t.mean[m].text);
          }
        }
        if (!backs.length && t.syn && t.syn.length) {
          for (var s = 0; s < Math.min(3, t.syn.length); s++) {
            if (t.syn[s] && t.syn[s].text && backs.indexOf(t.syn[s].text) === -1) backs.push(t.syn[s].text);
          }
        }
        out.push({ text: t.text, pos: pos, backs: backs });
      }
    }
    function walk(node, depth) {
      if (!node || depth > 6 || out.length >= dictMaxVariants) return;
      if (node.text && node.tr) { addEntry(node); return; }
      var keys = ['regular', 'synthetic'];
      for (var i = 0; i < keys.length; i++) {
        var arr = node[keys[i]];
        if (arr && arr.length) {
          for (var j = 0; j < arr.length && out.length < dictMaxVariants; j++) addEntry(arr[j]);
        }
      }
      for (var key in node) {
        if (Object.prototype.hasOwnProperty.call(node, key) && keys.indexOf(key) === -1) walk(node[key], depth + 1);
      }
    }
    if (json[pair]) walk(json[pair], 0); else walk(json, 0);
    return out;
  }

  function wrapFast(txt, len) {
    if (!txt || txt.length <= len) return txt.trim();
    var res = [], i = 0, j, k;
    while (i < txt.length) {
      j = i + len;
      if (j >= txt.length) { res.push(txt.substring(i).trim()); break; }
      k = txt.lastIndexOf(" ", j);
      if (k <= i) k = j;
      res.push(txt.substring(i, k).trim());
      i = k + 1;
    }
    return res.join("\n");
  }

  function wrapByPixels(text, font, maxWidth) {
    var cache = {};
    function measureW(s) {
      if (cache[s] === undefined) {
        cache[s] = Forms.TextRenderer.MeasureText(s, font, new Drawing.Size(10000, 0), Forms.TextFormatFlags.NoPadding).Width;
      }
      return cache[s];
    }
    var lines = [];
    var paragraphs = text.split("\n");
    for (var p = 0; p < paragraphs.length; p++) {
      var para = paragraphs[p];
      if (!para.trim()) { lines.push(""); continue; }
      var words = para.split(" ");
      var cur = "";
      for (var w = 0; w < words.length; w++) {
        var test = cur ? (cur + " " + words[w]) : words[w];
        if (measureW(test) > maxWidth && cur) {
          lines.push(cur);
          cur = words[w];
          if (measureW(cur) > maxWidth) {
            while (cur.length > 0) {
              var lo = 1, hi = cur.length;
              while (lo < hi) {
                var mid = Math.ceil((lo + hi) / 2);
                if (measureW(cur.substring(0, mid)) <= maxWidth) lo = mid; else hi = mid - 1;
              }
              lines.push(cur.substring(0, lo));
              cur = cur.substring(lo);
            }
            cur = "";
          }
        } else {
          cur = test;
        }
      }
      if (cur) lines.push(cur);
    }
    return lines;
  }

  function visualLen(s) {
    var w = 0;
    for (var i = 0; i < s.length; i++) {
      var c = s.charCodeAt(i);
      w += ((c>=0x1100&&c<=0x115F)||(c>=0x2E80&&c<=0xA4CF)||(c>=0xAC00&&c<=0xD7A3)||
            (c>=0xF900&&c<=0xFAFF)||(c>=0xFE30&&c<=0xFE4F)||(c>=0xFF00&&c<=0xFF60)||
            (c>=0xFFE0&&c<=0xFFE6)) ? 2 : 1;
    }
    return w;
  }

          if (cachedLines[ac].isFirst && cachedLines[ac].text.length > 0 && LETTER_RE.test(cachedLines[ac].text.charAt(0))) accentCount++;
        }
      }

      var gapCount = 0;
      if (enableAccent && accentMode === 'gap' && hasLong && accentCount > 1) {
        for (var gh = 0; gh < cachedLines.length; gh++) {
          if (gh > 0 && cachedLines[gh].isFirst && cachedLines[gh].text.length > 0 && LETTER_RE.test(cachedLines[gh].text.charAt(0))) gapCount++;
        }
      }
       var panelH = gapCount * paragraphGap;
      for (var ph = 0; ph < cachedLines.length; ph++) {
        panelH += cachedLines[ph].text ? LINE_H : paragraphGap;
      }           
      panel.Width = panelW;
      panel.Height = panelH;
      panel.Tag = { text: text, lines: cachedLines, color: tColor, hasLong: enableAccent && hasLong && accentCount > 1, accentMode: accentMode };

      panel.GetType().GetProperty("DoubleBuffered",
        host.flags(clr.System.Reflection.BindingFlags.NonPublic, clr.System.Reflection.BindingFlags.Instance))
        .SetValue(panel, true);

           panel.Paint.connect(function(s, e) {
        var g = e.Graphics;
        var d = s.Tag;
        var y = 0;
        for (var li = 0; li < d.lines.length; li++) {
          var ln = d.lines[li];
          if (!ln.text) { y += paragraphGap; continue; }
          if (d.accentMode === 'gap' && li > 0 && ln.isFirst && d.hasLong && ln.text.length > 0 && LETTER_RE.test(ln.text.charAt(0))) {
            y += paragraphGap;
          }
          Forms.TextRenderer.DrawText(g, ln.text, uiFont, new Drawing.Point(0, y), d.color, Forms.TextFormatFlags.NoPadding);
          if (d.accentMode !== 'gap' && ln.isFirst && d.hasLong && ln.text.length > 0 && LETTER_RE.test(ln.text.charAt(0))) {
            if (d.accentMode === 'color') {
              Forms.TextRenderer.DrawText(g, ln.text.substring(0, 1), uiFont, new Drawing.Point(0, y), COLOR_ACCENT, Forms.TextFormatFlags.NoPadding);
            } else {
              var charW = Forms.TextRenderer.MeasureText(ln.text.substring(0, 1), uiFont, new Drawing.Size(10000, 0), Forms.TextFormatFlags.NoPadding).Width;
              var pen = new Drawing.Pen(COLOR_ACCENT, 2);
              g.DrawLine(pen, 2, y + uiFont.Height - 1, charW - 13, y + uiFont.Height - 1);
              pen.Dispose();
            }
          }
          y += LINE_H;
        }
      });

      f.Controls.Add(panel);
      panel.BringToFront();
      attachDrag(panel);
      contentLabels.push(panel);
    }
  }

  function getContentSize() {
    var maxW = 0, totalH = 0;
    var sizes = [];
    for (var i = 0; i < contentLabels.length; i++) {
      var c = contentLabels[i];

      sizes.push({ w: w, h: h });

    if (totalH > 0) totalH -= 2;
    return { width: maxW, height: totalH, sizes: sizes };
  }

      if(x+f.Width>wa.Right)x=wa.Right-f.Width;

      if(x<wa.Left)x=wa.Left; if(y<wa.Top)y=wa.Top;
      f.Location = new Drawing.Point(x,y);
    } catch(_) {}
