function extractTextFromStructuredJson(json) {
try {
var data = JSON.parse(json);
if (!data || !data.blocks || data.error) return null;
var lines = [];
for (var b = 0; b < data.blocks.length; b++) {
var blk = data.blocks[b];
if (!blk || !blk.lines) continue;
for (var l = 0; l < blk.lines.length; l++) {
if (blk.lines[l] && blk.lines[l].text) lines.push(blk.lines[l].text);
}
}
return lines.join("\n");
} catch (e) {
sp.ConsoleError("extractTextFromStructuredJson: " + (e.message || e), "SS", Drawing.Color.Red);
return null;
}
}

function initLines() {
state.vLines = [];
var allGaps = [];
for (var r = 0; r < state.rows.length; r++) {
var ws = state.rows[r].words;
for (var k = 1; k < ws.length; k++) {
var g = ws[k].x - (ws[k-1].x + ws[k-1].w);
if (g > 0) allGaps.push(g);
}
}
var medGap = allGaps.length > 0 ? median(allGaps) : 5;
var minChannel = Math.max(3, medGap * 0.8);
var iv = [];
for (var r = 0; r < state.rows.length; r++) {
for (var k = 0; k < state.rows[r].words.length; k++) {
var w = state.rows[r].words[k];
iv.push({ l: w.x, r: w.x + w.w });
}
}
iv.sort(function(a, b) { return a.l - b.l; });
var cov = [];
for (var i = 0; i < iv.length; i++) {
if (cov.length === 0) { cov.push({ l: iv[i].l, r: iv[i].r }); continue; }
var last = cov[cov.length - 1];
if (iv[i].l <= last.r) { if (iv[i].r > last.r) last.r = iv[i].r; }
else cov.push({ l: iv[i].l, r: iv[i].r });
}
for (var i = 0; i < cov.length - 1; i++) {
var cl = cov[i].r, cr = cov[i + 1].l;
if (cr - cl >= minChannel) state.vLines.push((cl + cr) / 2);
}
}

function findNearestLineIdx(imgX, threshold) {

function findWordAt(imgX, imgY) {
if (imgX < 0 || imgY < 0 || !lastOcrBitmap || imgX > lastOcrBitmap.Width || imgY > lastOcrBitmap.Height) return -1;
var bestIdx = -1;
var bestDist = Infinity;
for (var i = 0; i < state.words.length; i++) {
var w = state.words[i];
if (isHiddenText(w.text)) continue;
var clickW = w._clickW || w.w;
if (imgX >= w.x && imgX <= w.x + clickW && imgY >= w.y && imgY <= w.y + w.h) {
var centerX = w.x + clickW / 2;
var centerY = w.y + w.h / 2;
var dist = Math.abs(imgX - centerX) + Math.abs(imgY - centerY);
if (dist < bestDist) {
bestDist = dist;
bestIdx = i;
}
}
}
return bestIdx;
}

function findWordFromPreview(e) {
if (!lastOcrBitmap || state.rows.length === 0) return null;
var charIdx = tbResult.GetCharIndexFromPosition(new Drawing.Point(e.X, e.Y));
var lineIdx = tbResult.GetLineFromCharIndex(charIdx);
if (lineIdx < 0 || lineIdx >= state.rows.length) return null;
var lineStart = tbResult.GetFirstCharIndexFromLine(lineIdx);
var posInLine = charIdx - lineStart;
var lineText = tbResult.Lines[lineIdx] || "";
if (lineText.length === 0) return null;
var rowWords = state.rows[lineIdx].words;
if (!rowWords || rowWords.length === 0) return null;
rowWords = rowWords.filter(function(w) {
    return !isHiddenText(w.text);
});
if (rowWords.length === 0) return null;
var bestWordIdx = -1;
var bestFoundIdx = 0;
var bestDist = Infinity;
var searchPos = 0;
for (var i = 0; i < rowWords.length; i++) {
var w = rowWords[i];
var foundIdx = lineText.indexOf(w.text, searchPos);
if (foundIdx < 0) continue;
var wordEnd = foundIdx + w.text.length;
if (posInLine >= foundIdx && posInLine <= wordEnd) {
return { wordIdx: state.words.indexOf(w), caretPos: Math.max(0, posInLine - foundIdx) };
}
var dist = Math.min(Math.abs(posInLine - foundIdx), Math.abs(posInLine - wordEnd));
if (dist < bestDist) {
bestDist = dist;
bestWordIdx = state.words.indexOf(w);
bestFoundIdx = foundIdx;
}
searchPos = wordEnd;
}
if (bestWordIdx >= 0) {
return { wordIdx: bestWordIdx, caretPos: Math.max(0, posInLine - bestFoundIdx) };
}
return null;
}

function updateResult(skipClipboard) {
var sorted = state.vLines.slice().sort(function(a, b) { return a - b; });
var colCount = sorted.length + 1;

function colOf(x) {
for (var c = 0; c < sorted.length; c++) if (x < sorted[c]) return c;
return colCount - 1;
}
var noSpaceGap = Math.max(2, state.analysis.medianHeight * 0.25);
var isSpaceJoining = state.analysis.isSpaceJoining;
var matrix = [];
for (var r = 0; r < state.rows.length; r++) {
var ws = state.rows[r].words;
ws = ws.filter(function(w) {
    return !isHiddenText(w.text);
});
if (!ws || ws.length === 0) continue;
var cells = [];
for (var c = 0; c < colCount; c++) cells.push("");
var curCol = colOf(ws[0].x + ws[0].w / 2);
var curTxt = ws[0].text;
for (var k = 1; k < ws.length; k++) {
var nc = colOf(ws[k].x + ws[k].w / 2);
var g = ws[k].x - (ws[k-1].x + ws[k-1].w);
if (nc === curCol) {
curTxt += (g < noSpaceGap || isSpaceJoining) ? "" : " ";
curTxt += ws[k].text;
} else {
cells[curCol] = cells[curCol] ? cells[curCol] + " " + curTxt : curTxt;
curCol = nc; curTxt = ws[k].text;
}
}
cells[curCol] = cells[curCol] ? cells[curCol] + " " + curTxt : curTxt;
matrix.push(cells);
}
state.lastMatrix = matrix;
var colWidths = [];
for (var c = 0; c < colCount; c++) {
var mw = 0;
for (var r = 0; r < matrix.length; r++) {
if (c < matrix[r].length) {
var w = displayWidth(matrix[r][c]);
if (w > mw) mw = w;
}
}
colWidths.push(mw);
}
var lines = [];
for (var r = 0; r < matrix.length; r++) {
var line = "";
for (var c = 0; c < matrix[r].length; c++) {
var cell = matrix[r][c] || "";
line += padToWidth(cell, colWidths[c]);
if (c < matrix[r].length - 1) line += "  ";
}
lines.push(line.replace(/\s+$/, ""));
}
tbResult.Text = lines.join("\r\n");
if (!skipClipboard) {
var tsv = [];
for (var r = 0; r < matrix.length; r++) tsv.push(matrix[r].join("\t"));
try { clip.SetText(tsv.join("\r\n")); } catch(e) { sp.ConsoleError("updateResult clipboard: " + (e.message || e), "SS", Drawing.Color.Red); }
}
}

function median(arr) {
if (!arr || arr.length === 0) return 0;
var s = arr.slice().sort(function(a, b) { return a - b; });
var m = Math.floor(s.length / 2);
return s.length % 2 === 0 ? (s[m - 1] + s[m]) / 2 : s[m];
}

function spread(arr) {
if (!arr || arr.length === 0) return 0;
var mn = arr[0], mx = arr[0];
for (var q = 1; q < arr.length; q++) { if (arr[q] < mn) mn = arr[q]; if (arr[q] > mx) mx = arr[q]; }
return mx - mn;
}

function minOf(arr) { var m = arr[0]; for (var q = 1; q < arr.length; q++) if (arr[q] < m) m = arr[q]; return m; }
function maxOf(arr) { var m = arr[0]; for (var q = 1; q < arr.length; q++) if (arr[q] > m) m = arr[q]; return m; }

function analyzeOcrWords(words) {
if (!words || words.length < 2) return null;
var heights = [], widths = [], sumW = 0, sumDisplayWidth = 0, sample = "";
for (var i = 0; i < words.length; i++) {
if (words[i].h > 0) heights.push(words[i].h);
if (words[i].w > 0) widths.push(words[i].w);
sumW += words[i].w;
sumDisplayWidth += displayWidth(words[i].text);
sample += words[i].text;
}
var medianHeight = median(heights);
if (!isFinite(medianHeight) || medianHeight <= 0) medianHeight = 20;
var medianWidth = median(widths);
if (!isFinite(medianWidth) || medianWidth <= 0) medianWidth = 40;
var isSpaceJoining = RE_SPACE_JOINING.test(sample);
return {
medianHeight: medianHeight,
medianWidth: medianWidth,
sumW: sumW,
sumDisplayWidth: sumDisplayWidth,
isSpaceJoining: isSpaceJoining
};
}

function clusterRowsByY(words, medianHeight) {
var rowCenterThreshold = Math.max(4, medianHeight * 0.75);
var byY = words.slice().sort(function(a, b) {
var ca = a.y + a.h / 2, cb = b.y + b.h / 2;
return (ca !== cb) ? ca - cb : a.x - b.x;
});
var rows = [];
for (var i = 0; i < byY.length; i++) {
var wb = byY[i], cy = wb.y + wb.h / 2;
if (rows.length === 0) {
rows.push({ top: wb.y, bottom: wb.y + wb.h, words: [wb] });
continue;
}
var last = rows[rows.length - 1], lc = (last.top + last.bottom) / 2;
if (Math.abs(cy - lc) <= rowCenterThreshold) {
last.words.push(wb);
last.top = Math.min(last.top, wb.y);
last.bottom = Math.max(last.bottom, wb.y + wb.h);
} else {
rows.push({ top: wb.y, bottom: wb.y + wb.h, words: [wb] });
}
}
var mergeThreshold = Math.max(rowCenterThreshold * 1.5, medianHeight);
var ri = 0;
while (ri < rows.length - 1) {
var cur = rows[ri], nxt = rows[ri + 1];
if (cur.words.length <= 1 && Math.abs((nxt.top + nxt.bottom) / 2 - (cur.top + cur.bottom) / 2) <= mergeThreshold) {
nxt.words = cur.words.concat(nxt.words);
nxt.top = Math.min(cur.top, nxt.top);
nxt.bottom = Math.max(cur.bottom, nxt.bottom);
rows.splice(ri, 1);
continue;
}
ri++;
}
ri = 1;
while (ri < rows.length) {
var prv = rows[ri - 1], cur = rows[ri];
if (cur.words.length <= 1 && Math.abs((cur.top + cur.bottom) / 2 - (prv.top + prv.bottom) / 2) <= mergeThreshold) {
prv.words = prv.words.concat(cur.words);
prv.top = Math.min(cur.top, prv.top);
prv.bottom = Math.max(cur.bottom, prv.bottom);
rows.splice(ri, 1);
continue;
}
ri++;
}
for (var r = 0; r < rows.length; r++) {
rows[r].words.sort(function(a, b) { return a.x - b.x; });
}
return rows;
}
var _displayWidthCache = new Map();

function displayWidth(s) {
    if (_displayWidthCache.has(s)) return _displayWidthCache.get(s); 
    
    var w = 0;
    for (var i = 0; i < s.length; i++) {
        var c = s.charCodeAt(i);
        var wide = (c >= 0x1100 && (c <= 0x115F || c === 0x2329 || c === 0x232A ||
            (c >= 0x2E80 && c <= 0xA4CF && c !== 0x303F) ||
            (c >= 0xAC00 && c <= 0xD7A3) ||
            (c >= 0xF900 && c <= 0xFAFF) ||
            (c >= 0xFE30 && c <= 0xFE4F) ||
            (c >= 0xFF00 && c <= 0xFF60) ||
            (c >= 0xFFE0 && c <= 0xFFE6)));
        w += wide ? 2 : 1;
    }
    _displayWidthCache.set(s, w);
    return w;
}

function padToWidth(s, target) {
var cur = displayWidth(s);
while (cur < target) { s += " "; cur++; }
return s;
}

function padStartWidth(s, target) {
var cur = displayWidth(s);
var pad = "";
while (cur < target) { pad += " "; cur++; }
return pad + s;
}

function formatAsOneLine(text) {
return text.replace(/\r?\n/g, " ").replace(/\s+/g, " ").trim();
}

function formatAsTable(text) {
var lines = text.split(RE_NL_SPLIT).filter(function(l) { return l.trim() !== ""; });
if (lines.length < 2) {
return text;
}
var matrix = [];
var maxCols = 0;
for (var i = 0; i < lines.length; i++) {
var cols = lines[i].split(RE_MULTI_SPACE).map(function(c) { return c.trim(); }).filter(function(c) { return c !== ""; });
matrix.push(cols);
if (cols.length > maxCols) maxCols = cols.length;
}
var allHaveMultipleCols = matrix.every(function(row) { return row.length >= 2; });
if (!allHaveMultipleCols || maxCols < 2) {
return text;
}
var colWidths = [];
for (var c = 0; c < maxCols; c++) {
var maxWidth = 0;
for (var r = 0; r < matrix.length; r++) {
if (matrix[r][c]) {
var cw = displayWidth(matrix[r][c]);
if (cw > maxWidth) maxWidth = cw;
}
}
colWidths.push(maxWidth);
}
var result = [];
for (var r = 0; r < matrix.length; r++) {
var rowParts = [];
for (var c = 0; c < maxCols; c++) {
var cell = matrix[r][c] || "";
var isNumber = RE_IS_NUMBER.test(cell.trim());
if (isNumber) {
rowParts.push(padStartWidth(cell, colWidths[c]));
} else {
rowParts.push(padToWidth(cell, colWidths[c]));
}
}
result.push(rowParts.join("  "));
}
var finalResult = result.join("\n");
return finalResult;
}

function formatAsTableTextGrab(fallbackText) {
var data = prepareTableData();
if (!data) return null;
var words = data.words;
var analysis = data.analysis;
var rowClusters = data.rows;
var medianHeight = analysis.medianHeight;
var resultLines = [];
var noSpaceThreshold = Math.max(2, medianHeight * 0.25);
var tabThreshold = Math.max(25, medianHeight * 4.5);
for (var r = 0; r < rowClusters.length; r++) {
var rc = rowClusters[r];
if (rc.words.length === 0) continue;
var lineStr = rc.words[0].text;
for (var i = 1; i < rc.words.length; i++) {
var prevWord = rc.words[i - 1];
var currWord = rc.words[i];
var gap = currWord.x - (prevWord.x + prevWord.w);
if (gap < noSpaceThreshold) {
lineStr += currWord.text;
} else if (gap >= tabThreshold) {
lineStr += "\t" + currWord.text;
} else {
lineStr += " " + currWord.text;
}
}
resultLines.push(lineStr);
}
return resultLines.join("\n");
}

function formatTableVisual(fallbackText) {
var data = prepareTableData();
if (!data) return null;
var words = data.words;
var analysis = data.analysis;
var rows = data.rows;
var medianHeight = analysis.medianHeight;
var sumW = analysis.sumW;
var sumDisplayWidth = analysis.sumDisplayWidth;
var isSpaceJoining = analysis.isSpaceJoining;
var noSpaceGap = Math.max(2, medianHeight * 0.25);
var allGaps = [];
for (var r = 0; r < rows.length; r++) {
var ws = rows[r].words, mg = 0;
for (var k = 1; k < ws.length; k++) {
var g = ws[k].x - (ws[k - 1].x + ws[k - 1].w);
allGaps.push(g);
if (g > mg) mg = g;
}
rows[r].maxGap = mg;
}
var medGap = median(allGaps);
var gapMin = Math.max(8, medGap * 1.6, medianHeight * 0.5);
for (var r = 0; r < rows.length; r++) {
rows[r].isTable = (rows[r].words.length >= 2 && rows[r].maxGap >= gapMin);
}
var iv = [];
for (var r = 0; r < rows.length; r++) {
if (!rows[r].isTable) continue;
for (var k = 0; k < rows[r].words.length; k++) {
var w2 = rows[r].words[k];
iv.push({ l: w2.x, r: w2.x + w2.w });
}
}
if (iv.length < 2) return null;
iv.sort(function (a, b) { return a.l - b.l; });
var cov = [];
for (var i = 0; i < iv.length; i++) {
if (cov.length === 0) { cov.push({ l: iv[i].l, r: iv[i].r }); continue; }
var lc2 = cov[cov.length - 1];
if (iv[i].l <= lc2.r) { if (iv[i].r > lc2.r) lc2.r = iv[i].r; }
else cov.push({ l: iv[i].l, r: iv[i].r });
}
var minChannel = Math.max(5, medGap * 1.4);
var seps = [];
for (var i = 0; i < cov.length - 1; i++) {
if (cov[i + 1].l - cov[i].r >= minChannel) seps.push({ l: cov[i].r, r: cov[i + 1].l });
}
if (seps.length === 0) return null;
var cols = [];
var startL = cov[0].l;
for (var i = 0; i < seps.length; i++) { cols.push({ l: startL, r: seps[i].l }); startL = seps[i].r; }
cols.push({ l: startL, r: cov[cov.length - 1].r });
var xOffset = cov[0].l;

function colOf(x) {
for (var c = 0; c < cols.length; c++) if (x < cols[c].r) return c;
return cols.length - 1;
}
var charW = (sumDisplayWidth > 0) ? (sumW / sumDisplayWidth) : medianHeight * 0.5;
if (!isFinite(charW) || charW <= 0) charW = medianHeight * 0.5;
if (charW < medianHeight * 0.30) charW = medianHeight * 0.30;
if (charW > medianHeight * 1.00) charW = medianHeight * 1.00;
for (var r = 0; r < rows.length; r++) {
var ws2 = rows[r].words, cells = [];
if (!rows[r].isTable) {
var txt = "", L = ws2[0].x, R = ws2[0].x + ws2[0].w;
for (var k = 0; k < ws2.length; k++) {
if (k > 0) {
var g2 = ws2[k].x - (ws2[k - 1].x + ws2[k - 1].w);
txt += (g2 < noSpaceGap || isSpaceJoining) ? "" : " ";
}
txt += ws2[k].text;
R = ws2[k].x + ws2[k].w;
}
cells.push({ text: txt, l: L, r: R, col: -1 });
} else {
var cCol = colOf(ws2[0].x + ws2[0].w / 2);
var cTxt = ws2[0].text, cL = ws2[0].x, cR = ws2[0].x + ws2[0].w;
for (var k = 1; k < ws2.length; k++) {
var nc = colOf(ws2[k].x + ws2[k].w / 2);
var g3 = ws2[k].x - (ws2[k - 1].x + ws2[k - 1].w);
if (nc === cCol) {
cTxt += (g3 < noSpaceGap || isSpaceJoining) ? "" : " ";
cTxt += ws2[k].text;
cR = ws2[k].x + ws2[k].w;
} else {
cells.push({ text: cTxt, l: cL, r: cR, col: cCol });
cCol = nc; cTxt = ws2[k].text; cL = ws2[k].x; cR = ws2[k].x + ws2[k].w;
}
}
cells.push({ text: cTxt, l: cL, r: cR, col: cCol });
}
rows[r].cells = cells;
}
for (var c = 0; c < cols.length; c++) {
var ls = [], rs = [], cs = [];
for (var r = 0; r < rows.length; r++) {
var cl = rows[r].cells;
for (var k = 0; k < cl.length; k++) {
if (cl[k].col === c) { ls.push(cl[k].l); rs.push(cl[k].r); cs.push((cl[k].l + cl[k].r) / 2); }
}
}
if (ls.length === 0) { cols[c].align = "left"; cols[c].charPos = 0; continue; }
var sL = spread(ls), sR = spread(rs), sC = spread(cs);
if (sL <= sR && sL <= sC)      { cols[c].align = "left";   cols[c].anchor = minOf(ls); }
else if (sR <= sC)             { cols[c].align = "right";  cols[c].anchor = maxOf(rs); }
else                           { cols[c].align = "center"; cols[c].anchor = (minOf(cs) + maxOf(cs)) / 2; }
cols[c].charPos = Math.round((cols[c].anchor - xOffset) / charW);
}
var placements = [], minStart = null;
for (var r = 0; r < rows.length; r++) {
var cl2 = rows[r].cells, rowPl = [];
for (var k = 0; k < cl2.length; k++) {
var cell = cl2[k];
var cellDW = displayWidth(cell.text);
var st;
if (cell.col < 0) {
st = Math.round((cell.l - xOffset) / charW);
} else if (cols[cell.col].align === "right") {
st = cols[cell.col].charPos - cellDW;
} else if (cols[cell.col].align === "center") {
st = cols[cell.col].charPos - Math.floor(cellDW / 2);
} else {
st = cols[cell.col].charPos;
}
if (st < 0) st = 0;
rowPl.push({ start: st, text: cell.text });
if (minStart === null || st < minStart) minStart = st;
}
placements.push(rowPl);
}
var shift = (minStart === null || minStart <= 0) ? 0 : minStart;
var out = [];
for (var r = 0; r < placements.length; r++) {
var pl = placements[r];
pl.sort(function (a, b) { return a.start - b.start; });
var line = "";
for (var k = 0; k < pl.length; k++) {
var t = pl[k].start - shift;
var currentDW = displayWidth(line);
if (t < currentDW) t = currentDW + 1;
while (displayWidth(line) < t) line += " ";
line += pl[k].text;
}
out.push(line.replace(/\s+$/, ""));
}
return out.join("\n");
}

function formatTableHard(fallbackText, gap) {
var data = prepareTableData();
if (!data) return null;
var words = data.words;
var analysis = data.analysis;
var rowClusters = data.rows;
var medianHeight = analysis.medianHeight;
var noSpaceThreshold = Math.max(2, medianHeight * 0.25);
var tabThreshold = Math.max(25, medianHeight * 4.5);
var rowTexts = [];
var isTableRow = [];
for (var r = 0; r < rowClusters.length; r++) {
var rc = rowClusters[r];
if (rc.words.length === 0) continue;
var maxGap = 0;
for (var i = 1; i < rc.words.length; i++) {
var gapVal = rc.words[i].x - (rc.words[i-1].x + rc.words[i-1].w);
if (gapVal > maxGap) maxGap = gapVal;
}
var isTable = (rc.words.length >= 2 && maxGap >= tabThreshold);
isTableRow.push(isTable);
var rowWords = [];
var currentWord = rc.words[0].text;
var currentWordX = rc.words[0].x;
var currentWordW = rc.words[0].w;
for (var i = 1; i < rc.words.length; i++) {
var prevWord = rc.words[i - 1];
var currWord = rc.words[i];
var gapVal = currWord.x - (prevWord.x + prevWord.w);
if (gapVal < noSpaceThreshold) {
currentWord += currWord.text;
currentWordW = (currWord.x + currWord.w) - currentWordX;
} else if (gapVal >= tabThreshold && isTable) {
rowWords.push({ text: currentWord, x: currentWordX, w: currentWordW });
currentWord = currWord.text;
currentWordX = currWord.x;
currentWordW = currWord.w;
} else {
currentWord += " " + currWord.text;
currentWordW = (currWord.x + currWord.w) - currentWordX;
}
}
rowWords.push({ text: currentWord, x: currentWordX, w: currentWordW });
rowTexts.push(rowWords);
}
var maxCols = 0;
for (var r = 0; r < rowTexts.length; r++) {
if (isTableRow[r] && rowTexts[r].length > maxCols) maxCols = rowTexts[r].length;
}
var colWidths = [];
for (var c = 0; c < maxCols; c++) {
var mw = 0;
for (var r = 0; r < rowTexts.length; r++) {
if (isTableRow[r] && c < rowTexts[r].length) {
var cw = displayWidth(rowTexts[r][c].text);
if (cw > mw) mw = cw;
}
}
colWidths.push(mw);
}
var sep = "";
for (var i = 0; i < gap; i++) sep += " ";
var resultLines = [];
for (var r = 0; r < rowTexts.length; r++) {
if (!isTableRow[r]) {
var lineStr = "";
for (var i = 0; i < rowTexts[r].length; i++) {
if (i > 0) lineStr += " ";
lineStr += rowTexts[r][i].text;
}
resultLines.push(lineStr);
} else {
var parts = [];
for (var c = 0; c < rowTexts[r].length; c++) {
var cell = rowTexts[r][c].text;
if (c < rowTexts[r].length - 1 && c < maxCols) {
cell = padToWidth(cell, colWidths[c]);
}
parts.push(cell);
}
resultLines.push(parts.join(sep));
}
}
return resultLines.join("\n");
}

function createRotatedBitmap(angle) {
if (!rotationPaddedBitmap) return null;
angle = normalizeAngle(angle);
var srcW = rotationPaddedBitmap.Width;
var srcH = rotationPaddedBitmap.Height;
var radians = angle * Math.PI / 180;
var cos = Math.abs(Math.cos(radians));
var sin = Math.abs(Math.sin(radians));
var newW = Math.floor(srcW * cos + srcH * sin);
var newH = Math.floor(srcW * sin + srcH * cos);
if (newW < 1) newW = 1;
if (newH < 1) newH = 1;
var bmp = new Drawing.Bitmap(toInt(newW), toInt(newH), Drawing.Imaging.PixelFormat.Format32bppPArgb);
var g = Drawing.Graphics.FromImage(bmp);
try {
g.Clear(Drawing.Color.Transparent);
g.InterpolationMode = Drawing.Drawing2D.InterpolationMode.HighQualityBicubic;
g.SmoothingMode = Drawing.Drawing2D.SmoothingMode.HighQuality;
g.PixelOffsetMode = Drawing.Drawing2D.PixelOffsetMode.HighQuality;
g.CompositingQuality = Drawing.Drawing2D.CompositingQuality.HighQuality;
g.CompositingMode = Drawing.Drawing2D.CompositingMode.SourceOver;
g.TranslateTransform(toFloat(newW / 2), toFloat(newH / 2));
g.RotateTransform(toFloat(angle));
g.TranslateTransform(toFloat(-srcW / 2), toFloat(-srcH / 2));
if (!_cachedImageAttr) {
_cachedImageAttr = new Drawing.Imaging.ImageAttributes();
_cachedImageAttr.SetWrapMode(Drawing.Drawing2D.WrapMode.TileFlipXY, Drawing.Color.Transparent);
}
var destRect = new Drawing.Rectangle(0, 0, toInt(srcW), toInt(srcH));
g.DrawImage(rotationPaddedBitmap, destRect, 0, 0, toInt(srcW), toInt(srcH), Drawing.GraphicsUnit.Pixel, _cachedImageAttr);
} finally { g.Dispose(); }
return bmp;
}

var oldImg = globalActivePic.Image;
globalActivePic.Image = bmp;
if (oldImg && oldImg !== rotationSourceBitmap && oldImg !== rotationPaddedBitmap) {
try { oldImg.Dispose(); } catch(e) {}
}
}

function sceneCenter(form) {
return {
cx: form.ClientSize.Width / 2,
cy: form.ClientSize.Height / 2
};
}

function computeSide() {
var srcW = rotationSourceBitmap.Width;
var srcH = rotationSourceBitmap.Height;
var diag = Math.sqrt(srcW * srcW + srcH * srcH);
return toInt(Math.ceil(diag * zoomFactor + 60));
}

function maxAllowedSide() {
try {
var form = globalPreviewForm;
if (form && !form.IsDisposed) {
var screen = Forms.Screen.FromControl(form);
if (screen && screen.Bounds) {
return Math.min(screen.Bounds.Width, screen.Bounds.Height) - 20;
// return Math.max(Number(screen.Bounds.Width), Number(screen.Bounds.Height)) - 20; // There is no zoom limit
}
}
} catch(e) {}
return 2500;
}

function resizeFormForZoom() {
var form = globalPreviewForm;
if (!form || form.IsDisposed || !rotationSourceBitmap) return;
var oldCx = Number(form.Left) + Number(form.Width) / 2;
var oldCy = Number(form.Top) + Number(form.Height) / 2;
var side = computeSide();
var maxSide = maxAllowedSide();
if (side > maxSide) side = maxSide;
currentSide = side;
form.Size = new Drawing.Size(side, side);
var actual = Number(form.Width);
if (actual > 10 && actual !== side) currentSide = actual;
form.Location = new Drawing.Point(toInt(oldCx - currentSide / 2), toInt(oldCy - currentSide / 2));
}

function applyGeometry() {
var form = globalPreviewForm;
if (!form || form.IsDisposed || !rotationSourceBitmap) return;
var cacheKey = roundInt(currentRotationAngle * 100) / 100;
if (lastRegionAngle === cacheKey &&
lastRegionZoom === zoomFactor &&
lastRotationHandleVisible === rotationHandleVisible &&
!isRotating) {
var c = sceneCenter(form);
var srcW = rotationSourceBitmap.Width;
var srcH = rotationSourceBitmap.Height;
var z = zoomFactor;
var radians = currentRotationAngle * Math.PI / 180;
var cosR = Math.cos(radians);
var sinR = Math.sin(radians);
var halfW = srcW / 2, halfH = srcH / 2;
var xr = halfW * cosR - halfH * sinR + c.cx;
var yr = halfW * sinR + halfH * cosR + c.cy;
var c2 = { x: c.cx + (xr - c.cx) * z, y: c.cy + (yr - c.cy) * z };
var ddx = c2.x - c.cx, ddy = c2.y - c.cy;
var dist = Math.sqrt(ddx * ddx + ddy * ddy);
var hx = dist > 0.001 ? c2.x - (ddx / dist) * HANDLE_INSET : c2.x - HANDLE_INSET;
var hy = dist > 0.001 ? c2.y - (ddy / dist) * HANDLE_INSET : c2.y - HANDLE_INSET;
rotationHandlePoint = new Drawing.Point(toInt(roundInt(hx)), toInt(roundInt(hy)));
return;
}
lastRegionAngle = cacheKey;
lastRegionZoom = zoomFactor;
lastRotationHandleVisible = rotationHandleVisible;
var c = sceneCenter(form);
var srcW = Number(rotationSourceBitmap.Width);
var srcH = Number(rotationSourceBitmap.Height);
var z = zoomFactor;
var radians = currentRotationAngle * Math.PI / 180;
var cosR = Math.cos(radians);
var sinR = Math.sin(radians);
var halfW = srcW / 2, halfH = srcH / 2;

function zp(x, y) {
var dx = x - halfW, dy = y - halfH;
var xr = dx * cosR - dy * sinR + c.cx;
var yr = dx * sinR + dy * cosR + c.cy;
return { x: c.cx + (xr - c.cx) * z, y: c.cy + (yr - c.cy) * z };
}
var q0 = zp(-2, -2);
var q1 = zp(srcW + 2, -2);
var q2 = zp(srcW + 2, srcH + 2);
var q3 = zp(-2, srcH + 2);
var dx = q2.x - c.cx;
var dy = q2.y - c.cy;
var dist = Math.sqrt(dx * dx + dy * dy);
var hx, hy;
if (dist > 0.001) {
hx = q2.x - (dx / dist) * HANDLE_INSET;
hy = q2.y - (dy / dist) * HANDLE_INSET;
} else {
hx = q2.x - HANDLE_INSET;
hy = q2.y - HANDLE_INSET;
}
rotationHandlePoint = new Drawing.Point(toInt(roundInt(hx)), toInt(roundInt(hy)));
var path = new Drawing.Drawing2D.GraphicsPath();
try {
path.FillMode = Drawing.Drawing2D.FillMode.Winding;
path.StartFigure();
path.AddLine(toInt(roundInt(q0.x)), toInt(roundInt(q0.y)), toInt(roundInt(q1.x)), toInt(roundInt(q1.y)));
path.AddLine(toInt(roundInt(q1.x)), toInt(roundInt(q1.y)), toInt(roundInt(q2.x)), toInt(roundInt(q2.y)));
path.AddLine(toInt(roundInt(q2.x)), toInt(roundInt(q2.y)), toInt(roundInt(q3.x)), toInt(roundInt(q3.y)));
path.AddLine(toInt(roundInt(q3.x)), toInt(roundInt(q3.y)), toInt(roundInt(q0.x)), toInt(roundInt(q0.y)));
path.CloseFigure();
if (rotationHandleVisible || isRotating) {
var rr = ROTATION_HANDLE_RADIUS + 2;
path.AddEllipse(new Drawing.Rectangle(
toInt(roundInt(hx - rr)), toInt(roundInt(hy - rr)),
toInt(rr * 2), toInt(rr * 2)));
}
var oldRegion = form.Region;
form.Region = new Drawing.Region(path);
if (oldRegion) { try { oldRegion.Dispose(); } catch(e) {} }
} finally { path.Dispose(); }
form.Refresh();
}

function isNearRotationHandle(x, y) {
if (!rotationHandlePoint) return false;
var dx = Number(x) - Number(rotationHandlePoint.X);
var dy = Number(y) - Number(rotationHandlePoint.Y);
return Math.sqrt(dx * dx + dy * dy) <= ROTATION_HANDLE_HIT_RADIUS;
}

function ensureCachedDrawingObjects() {
if (!_cachedHandleOutlinePen) {
_cachedHandleOutlinePen = new Drawing.Pen(Drawing.Color.Black, 2);
_cachedHandleArrowPen = new Drawing.Pen(Drawing.Color.White, 2);
_cachedHandleNormalBrush = new Drawing.SolidBrush(Drawing.Color.FromArgb(255, 90, 90, 90));
_cachedHandleActiveBrush = new Drawing.SolidBrush(Drawing.Color.FromArgb(255, 0, 120, 215));
}
}

function drawRotationHandle(g, isActive) {
if (!rotationHandlePoint) return;
ensureCachedDrawingObjects();
var hx = Number(rotationHandlePoint.X);
var hy = Number(rotationHandlePoint.Y);
var r = ROTATION_HANDLE_RADIUS;
try {
g.SmoothingMode = Drawing.Drawing2D.SmoothingMode.AntiAlias;
g.DrawEllipse(_cachedHandleOutlinePen, new Drawing.Rectangle(toInt(hx - r - 1), toInt(hy - r - 1), toInt((r + 1) * 2), toInt((r + 1) * 2)));
g.FillEllipse(isActive ? _cachedHandleActiveBrush : _cachedHandleNormalBrush, new Drawing.Rectangle(toInt(hx - r), toInt(hy - r), toInt(r * 2), toInt(r * 2)));
var innerR = r - 4;
if (innerR > 2) {
var innerRect = new Drawing.Rectangle(toInt(hx - innerR), toInt(hy - innerR), toInt(innerR * 2), toInt(innerR * 2));
g.DrawArc(_cachedHandleArrowPen, innerRect, 45, 270);
var endAngle = 315;
var endRad = endAngle * Math.PI / 180;
var px = hx + innerR * Math.cos(endRad);
var py = hy + innerR * Math.sin(endRad);
var len = innerR * 0.8;
var a1 = (endAngle + 150) * Math.PI / 180;
var a2 = (endAngle - 150) * Math.PI / 180;
g.DrawLine(_cachedHandleArrowPen, toInt(px), toInt(py), toInt(px + len * Math.cos(a1)), toInt(py + len * Math.sin(a1)));
g.DrawLine(_cachedHandleArrowPen, toInt(px), toInt(py), toInt(px + len * Math.cos(a2)), toInt(py + len * Math.sin(a2)));
}
} catch (e) {}
}

var form_mouseMove = form.MouseMove.connect(function (sender, e) {

lastMoveTime = now;
if (!form.Capture) form.Capture = true;
var mouseScreen = form.PointToScreen(new Point(e.X, e.Y));
var currentMouseAngle = Math.atan2(
Number(mouseScreen.Y) - rotationCenterScreenY,
Number(mouseScreen.X) - rotationCenterScreenX
) * 180 / Math.PI;
var delta = currentMouseAngle - rotationLastMouseAngle;
if (delta > 180) delta -= 360;
if (delta < -180) delta += 360;
var newAngle = currentRotationAngle + delta;
// MAGNETIC STICKING with "stop for a moment"
var snapped = snapTo90(newAngle);
if (snapped !== newAngle) {
if (!isSnapped) {
// We've just entered the zone, and we're stuck at 90°
isSnapped = true;
currentRotationAngle = snapped;
} else {
var accumulated = currentMouseAngle - rotationLastMouseAngle;
if (accumulated > 180) accumulated -= 360;
if (accumulated < -180) accumulated += 360;
if (Math.abs(accumulated) > SNAP_RELEASE_THRESHOLD) {
isSnapped = false;
currentRotationAngle = newAngle;
rotationLastMouseAngle = currentMouseAngle;
}
}
} else {
// Outside the sticking area — free rotation
isSnapped = false;
currentRotationAngle = newAngle;
rotationLastMouseAngle = currentMouseAngle;
}
markInteraction();
applyGeometry();
return;
}
if (isDragging) {
var mouseScreenDrag = form.PointToScreen(new Point(e.X, e.Y));
var newLeftNum = mouseScreenDrag.X - dragOffsetX;
var newTopNum = mouseScreenDrag.Y - dragOffsetY;

dragMoved = true;
}

form.Location = new Point(toInt(newLeftNum), toInt(newTopNum));
return;
}

rotationHandleVisible = nearHandle;
